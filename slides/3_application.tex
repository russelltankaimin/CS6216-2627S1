\section{Applications to Multi-Armed Bandits}

\begin{frame}{Multi-Armed Bandits Formulation}
    \begin{itemize}
        \item At round $t = 1, \dots, T$, learner select one of $d$ arms \pause
        \item $p^{(t)}$ is a r.v. denoting the learner's choice of arm \pause
        \item $y^{(t)} \in [0, 1]^d$ denotes the cost of choosing each arm at round $t$ \pause
        \item $y^{(t)}_{p^{(t)}} \in [0, 1]$ (also a r.v.) denotes the cost of choosing arm $p^{(t)}$ \pause
        \item Minimise $$ \mathbb{E}\left[\sum_{t=1}^T f^{(t)}(x^{(t)}) - f^{(t)}(x) \right] $$
        expectation is taken over the learner's randomness, $f^{(t)}$ is the cost function at round $t$
    \end{itemize}
\end{frame}

\begin{frame}{Multi-Armed Bandits (Linear Cost)}
    \begin{itemize}
        \item At round $t = 1, \dots, T$, learner select one of $d$ arms 
        \item $p^{(t)}$ is a r.v. denoting the learner's choice of arm
        \item $y^{(t)} \in [0, 1]^d$ denotes the cost of choosing each arm at round $t$
        \item $y^{(t)}_{p^{(t)}} \in [0, 1]$ (also a r.v.) denotes the cost of choosing arm $p^{(t)}$
        \item Minimise $$ \mathbb{E}\left[ \sum_{t=1}^T y^{(t)}_{p^{(t)}}\right] - \min_{i} \sum_{t=1}^T y^{(t)}_i $$
        expectation is taken over the learner's randomness and $f^{(t)}(x) = \langle x, y^{(t)} \rangle$ \pause
        \item Approach: Estimate the gradient/sub-gradient at each round and apply OMD
    \end{itemize}
\end{frame}

\begin{frame}{OMD with Estimated Gradients}
    \small
    \begin{block}{Regret of OMD with Estimated Gradients}
        Suppose that the $\texttt{OMD with Est. Gradients}$ algorithm is run on a sequence of loss functions $f^{(1)}, f^{(2)}, \dots, f^{(T)}$. Suppose that the estimated sub-gradients are chosen such that with probability 1 we have
        $$ \sum_{t=1}^T \langle x^{(t)} - x, z^{(t)} \rangle \leq B(x) + \sum_{t=1}^{T} ||z_t||^2_t,$$
        where $B$ is some function and for all round $t$ the norm $||\cdot||_t$ may depend on $x^{(t)}$. Then,
        $$ \mathbb{E}\left[ \sum_{t=1}^{T} \left( f^{(t)}(x^{(t)}) - f^{(t)}(x) \right) \right] \leq B(x) + \sum_{t=1}^{T} ||z_t||^2_t, $$
        where the expectation is taken with respect to the randomness in choosing $z^{(1)}, \dots, z^{(T)}$.
    \end{block}
\end{frame}

\begin{frame}{OMD with Estimated Gradients}
    Proof sketch:
    \begin{itemize}
        \item Apply expectation (w.r.t $z^{(1)}, \dots, z^{(T)}$) to conclude 
        $$ \mathbb{E}\left[\sum_{t=1}^T \langle x^{(t)} - x, z^{(t)} \rangle\right] = \mathbb{E}\left[\sum_{t=1}^T \langle x^{(t)} - x, v^{(t)} \rangle\right], $$
        where $v^{(t)} = \mathbb{E}\left[z^{(t)}|z^{(t-1)}, \dots, z^{(1)}\right]$.
        \item Apply definition of sub-gradient to conclude 
        $$ \langle x^{(t)} - x, v^{(t)} \rangle \geq f^{(t)}(x^{(t)}) - f^{(t)}(x). $$
    \end{itemize}
\end{frame}

        \item $y^{(t)} \in [0, 1]^d$ denotes the cost of choosing each arm at round $t$

        \item Minimise $$ \mathbb{E}\left[ \sum_{t=1}^T y^{(t)}_{p^{(t)}}\right] - \min_{i} \sum_{t=1}^T y^{(t)}_i $$
        expectation is taken over the learner's randomness and $f^{(t)}(x) = \langle x, y^{(t)} \rangle$ \pause
        \item Approach: Estimate the gradient/sub-gradient at each round and apply OMD
    \end{itemize}
\end{frame}

\begin{frame}{Multi-Armed Bandits (Linear Cost)}
    \begin{itemize}
        \item Main difficulty: If the learner chooses $p^{(t)}$, we only know $y^{(t)}_{p^{(t)}}$. We have no information on the rest of $y^{(t)}$
        \item Define the random variable $z^{(t)}$ such that
        $$
        z^{(t)}_j =
        \begin{cases}
        y^{(t)}_j / x^{(t)}_j & \text{if } j = p^{(t)} \\
        0 & \text{otherwise}
        \end{cases}
        $$
        where $x^{(t)}_j = P(p^{(t)} = j \ | \ z^{(t-1)}, \dots, z^{(1)})$
        \item $z^{(t)}$ is an unbiased estimator of $\nabla f^{(t)} = y^{(t)}$:
        \begin{align*}
            \mathbb{E}\left[ z^{(t)}_j | z^{(t-1)}, \dots, z^{(1)} \right] &= \sum_{i=1}^d z^{(t)}_j P(p^{(t)} = i) \\
            &= y^{(t)}_j / x^{(t)}_j \times x^{(t)}_j = y^{(t)}_j
        \end{align*}
    \end{itemize}
\end{frame}

\texttt{Multi-Armed Bandit Algorithm (Linear Cost)}

\begin{frame}{Multi-Armed Bandits (Linear Cost)}
    \small
    \begin{block}{Regret bound of Multi-Armed Bandit Algorithm (Linear Cost)}
        $$
        \mathbb{E}\left[\sum_{t=1}^{T} y^{(t)}_{p^{(t)}}\right] - \min_i \sum_{t=1}^{T} y^{(t)}_i
        \leq \frac{\log d}{\eta} + \eta d T.
        $$
        In particular, setting $\eta = \sqrt{\log(d)/(dT)}$ we obtain the regret bound of
        $2\sqrt{d \log(d) T}$.
    \end{block}\pause
    Proof strategy: We want to apply the following result
    \begin{block}{Local Norm Regret Bound}
        Assume that the normalised EG algorithm is run on a sequence of linear loss functions $f^{(t)}(x) = \langle x, z^{(t)} \rangle$ such that for all $t \in [T]$, $i \in [d]$, we have $\eta z^{(t)}_i \geq -1$. Then,
        $$
        \sum_{t=1}^T \langle x^{(t)} - x, z^{(t)} \rangle
        \leq \frac{\log(d)}{\eta} + \eta \sum_{t=1}^T ||z^{(t)}||^2_t.
        $$
    \end{block}
\end{frame}

    \end{align*}

\begin{frame}{Multi-Armed Bandits (Black-box Convex Cost)}
    \begin{itemize}
        \item Main difficulty 1: If the learner chooses $p^{(t)}$, we only know $y^{(t)}_{p^{(t)}}$. We have no information on the rest of $y^{(t)}$ \pause
        \item Main difficulty 2: $f$ may not even be differentiable \pause
        \item Idea: Construct a $\hat{f}$ that serves as the "smoothed" version of $f$, i.e.
        $$ \hat{f}(x) = \mathbb{E}_{v \sim U_b} [f(x + \delta v)] $$ \pause
        \item $f$ and $\hat{f}$ are close. If $f$ is a $L$-Lipschitz function, then $|\hat{f}(x) - f(x)| \leq L \delta$ \pause
        \item $\hat{f}$ is differentiable and 
        $$\mathbb{E}_{v \sim U_{sp}}\left[ \frac{d}{\delta}f(x + \delta v)v \right] = \nabla \hat{f}(x)$$
    \end{itemize}
\end{frame}

\begin{frame}{Multi-Armed Bandits (Black-box Convex Cost)}
\small
\begin{block}{Regret Bound for Multi-Armed Bandits (Black-box Convex Cost)}
    Consider running the general algorithm on a
    sequence $f^{(1)}, \ldots, f^{(T)}$ of $L$-Lipschitz convex functions. Let $S$ be a
    convex set and define $B = \max_{u \in S} \|u\|$ and
    $F = \max_{u \in S,\, t \in [T]} f^{(t)}(u)$. Then, for all $u \in S$ we have
    $$
    \mathbb{E}\left[\sum_t \left(f^{(t)}(x^{(t)} + \delta v^{(t)}) - f^{(t)}(u)\right)\right]
    \leq 3L\delta T + \frac{B^2}{2\eta} + \eta T d^2 (F/\delta + L)^2.
    $$
    In particular, setting $\eta = \frac{B}{d(F/\delta + L)\sqrt{2T}}$,
    $\delta = \sqrt{\frac{BdF}{3L}}\, T^{-1/4}$ we obtain that the regret is bounded
    by $O\big(\sqrt{BdFL}\, T^{3/4}\big)$.
\end{block}
\end{frame}

\begin{frame}{Multi-Armed Bandits (Black-box Convex Cost)}
\small
\begin{block}{Regret Bound for Multi-Armed Bandits (Black-box Convex Cost)}
    Consider running the general algorithm on a
    sequence $f^{(1)}, \ldots, f^{(T)}$ of $L$-Lipschitz convex functions. Let $S$ be a
    convex set and define $B = \max_{u \in S} \|u\|$ and
    $F = \max_{u \in S,\, t \in [T]} f^{(t)}(u)$. Then, for all $u \in S$ we have
    $$
    \mathbb{E}\left[\sum_t \left(f^{(t)}(x^{(t)} + \delta v^{(t)}) - f^{(t)}(u)\right)\right]
    \leq 3L\delta T + \frac{B^2}{2\eta} + \eta T d^2 (F/\delta + L)^2.
    $$
\end{block}
Proof sketch:
By convexity of $f^{(t)}$ (linearization trick), we have
\begin{align*}
